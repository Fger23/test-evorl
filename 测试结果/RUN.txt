# ============================================================
# RTC / CFG 启动命令
#
# 使用顺序：
# 1. 先在 H200 上启动“服务器”。
# 2. 再在机器人电脑上，从下面四种客户端模式中选择一种启动。
# 3. 同一时间只启动一个客户端，避免两个进程同时占用机械臂串口和相机。
# 4. 启动前确认 /dev/ttyACM*、/dev/video* 与实际设备一致。
#
# 固定参数：
# - 机器人控制频率：30 Hz
# - RTC：inference_delay=21，execution_horizon=35
# - RTC guidance weight=10.0，schedule=LINEAR
# - beta：所有客户端命令均显式填写为 1.5
# - CFG=false 时 beta 不参与计算；保留参数只是为了让命令配置完整、统一。
# ============================================================

# RTC 开关：
#
# Record 模式开启 RTC：
#     --remote_policy.rtc_enable=true \
#     --remote_policy.rtc_inference_delay=21 \
#     --remote_policy.rtc_execution_horizon=35 \
#
# Record 模式关闭 RTC（用这一行替换上面三行）：
#     --remote_policy.rtc_enable=false \
#
# robot_client 开启 RTC：
#     --rtc_enable=true \
#     --rtc_inference_delay=21 \
#     --rtc_execution_horizon=35 \
#
# robot_client 关闭 RTC（用这一行替换上面三行）：
#     --rtc_enable=false \
#
# CFG 和 RTC 可以独立组合：
# - CFG=false, RTC=true ：C-only + RTC
# - CFG=true,  RTC=true ：batch=2 CFG + RTC
# - CFG=true,  RTC=false：batch=2 CFG，不使用 RTC
# - CFG=false, RTC=false：普通单分支推理，不使用 CFG/RTC
#
# RTC=false 时 inference_delay/execution_horizon 不生效，应从命令中删除。
# beta 始终显式保留为 1.5；CFG=false 时 beta 不生效。


# ============================================================
# 一、H200 服务器
#
# 服务器只开启 ACP/RTC 能力；推理模式和 RTC 参数由客户端提供。
# ============================================================
python -m lerobot.async_inference.policy_server \
    --host=0.0.0.0 \
    --port=8090 \
    --acp_inference.enable=true


# ============================================================
# 二、Record 模式：CFG 关闭，C-only RTC
#
# 推理：batch=1，只计算 task + Advantage: positive。
# 队列：保存 C_raw / C_processed，下一次 leftover 来自 C_raw。
# beta=1.5 显式填写，但 CFG=false 时不会参与计算。
# ============================================================
python -m lerobot.scripts.lerobot_record \
    --robot.type=bi_so_follower \
    --robot.left_arm_config.port=/dev/ttyACM0 \
    --robot.right_arm_config.port=/dev/ttyACM1 \
    --robot.id=bimanual_follower \
    --robot.left_arm_config.cameras="{fixed: {type: intelrealsense, serial_number_or_name: '260322276426', width: 640, height: 480, fps: 30}, wrist: {type: opencv, index_or_path: /dev/v4l/by-path/pci-0000:c6:00.3-usb-0:2:1.0-video-index0, width: 640, height: 480, fps: 30, fourcc: MJPG}}" \
    --robot.right_arm_config.cameras="{wrist: {type: opencv, index_or_path: /dev/v4l/by-path/pci-0000:c6:00.3-usb-0:1:1.0-video-index0, width: 640, height: 480, fps: 30, fourcc: MJPG}}" \
    --remote_policy.enable=true \
    --remote_policy.server_address=10.61.5.8:8090 \
    --remote_policy.policy_type=pi05 \
    --remote_policy.pretrained_name_or_path=/nfs/lerobot/s101/models/evo/so101/policy/banknote_binding_iteration_1_pi05/checkpoints/last/pretrained_model \
    --remote_policy.policy_device=cuda:0 \
    --remote_policy.actions_per_chunk=50 \
    --remote_policy.obs_queue_timeout_s=30 \
    --remote_policy.chunk_size_threshold=0.7 \
    --remote_policy.rtc_enable=true \
    --remote_policy.use_cfg=false \
    --remote_policy.rtc_inference_delay=21 \
    --remote_policy.rtc_execution_horizon=35 \
    --remote_policy.rtc_cfg_beta=1.5 \
    --dataset.repo_id=local/tmp \
    --dataset.single_task="Pick up the banknotes from the desktop, straighten them out, and then place them into the banknote binding machine." \
    --dataset.video=false \
    --dataset.push_to_hub=false \
    --dataset.num_episodes=10 \
    --dataset.episode_time_s=3600 \
    --dataset.reset_time_s=15


# ============================================================

    --remote_policy.policy_device=cuda:0 \
